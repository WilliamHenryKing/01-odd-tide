async (page) => {
  const base = 'http://127.0.0.1:4511';
  const captures = [];
  await page.setViewportSize({ width: 1440, height: 1000 });
  for (const [name, hour] of [['day', 9], ['apricot', 18], ['night', 21]]) {
    await page.goto(`${base}/?time=${hour}`);
    await page.locator('.island-surface.ready').waitFor();
    await page.getByRole('button', { name: 'Gentle motion off', exact: true }).click();
    await page.getByRole('button', { name: 'Gentle motion on', exact: true }).waitFor();
    await page.waitForFunction(() => getComputedStyle(document.querySelector('.world-canvas')).opacity === '1');
    await page.screenshot({ path: `output/playwright/study-${name}.png` });
    if (name !== 'apricot') await page.getByLabel('Interactive miniature of the Odd Tide island', { exact: true }).screenshot({ path: `public/plates/island-${name}.jpg`, type: 'jpeg', quality: 88 });
    captures.push(name);
  }
  for (const id of ['weather-house', 'nap-observatory', 'lantern-lodge']) {
    await page.goto(`${base}/stays/${id}?time=9`);
    await page.locator('.island-surface.ready').waitFor();
    await page.getByRole('button', { name: 'Gentle motion off', exact: true }).click();
    await page.getByRole('button', { name: 'Gentle motion on', exact: true }).waitFor();
    await page.waitForFunction(() => getComputedStyle(document.querySelector('.world-canvas')).opacity === '1');
    await page.getByLabel('Interactive miniature of the Odd Tide island', { exact: true }).screenshot({ path: `public/plates/${id}-day.jpg`, type: 'jpeg', quality: 88 });
    await page.getByRole('button', { name: 'Take a peek inside', exact: true }).click();
    await page.getByRole('button', { name: 'Tuck the roof back in', exact: true }).waitFor();
    await page.screenshot({ path: `output/playwright/${id}-interior.png` });
    captures.push(id);
  }
  await page.setViewportSize({ width: 390, height: 844 });
  for (const [name, hour] of [['day', 9], ['night', 21]]) {
    await page.goto(`${base}/?time=${hour}`);
    await page.locator('.island-surface.ready').waitFor();
    await page.getByRole('button', { name: 'Gentle motion off', exact: true }).click();
    await page.getByRole('button', { name: 'Gentle motion on', exact: true }).waitFor();
    await page.waitForFunction(() => getComputedStyle(document.querySelector('.world-canvas')).opacity === '1');
    await page.getByLabel('Interactive miniature of the Odd Tide island', { exact: true }).screenshot({ path: `public/plates/island-${name}-phone.jpg`, type: 'jpeg', quality: 88 });
    await page.evaluate(() => window.scrollTo(0, 0));
    await page.screenshot({ path: `output/playwright/phone-${name}-final.png` });
  }
  return { captures, note: 'Captured actual WebGL output; lighting and closeup images still need visual inspection.' };
}
